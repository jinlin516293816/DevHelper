2. 大文件上传 ​

分片上传原理 ​

  1. 将大文件分割成多个小分片
  2. 逐个上传分片
  3. 服务器端合并分片

实现代码 ​

javascript
async function uploadLargeFile(file) {
  const chunkSize = 1024 * 1024; // 1MB 分片
  const totalChunks = Math.ceil(file.size / chunkSize);
  const fileId = generateUniqueId(); // 生成唯一文件ID
  
  console.log(`文件大小: ${file.size} bytes`);
  console.log(`分片数量: ${totalChunks}`);
  
  for (let i = 0; i < totalChunks; i++) {
    const start = i * chunkSize;
    const end = Math.min(start + chunkSize, file.size);
    const chunk = file.slice(start, end);
    
    const formData = new FormData();
    formData.append('file', chunk);
    formData.append('fileId', fileId);
    formData.append('chunkIndex', i);
    formData.append('totalChunks', totalChunks);
    formData.append('fileName', file.name);
    
    try {
      const response = await fetch('/upload-chunk', {
        method: 'POST',
        body: formData
      });
      
      if (!response.ok) {
        throw new Error(`上传分片 ${i} 失败`);
      }
      
      console.log(`分片 ${i + 1}/${totalChunks} 上传成功`);
    } catch (error) {
      console.error('上传分片失败:', error);
      return false;
    }
  }
  
  // 所有分片上传完成后，通知服务器合并
  try {
    const response = await fetch('/merge-chunks', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        fileId,
        fileName: file.name,
        totalChunks
      })
    });
    
    if (response.ok) {
      const result = await response.json();
      console.log('文件上传完成:', result);
      return true;
    } else {
      throw new Error('合并分片失败');
    }
  } catch (error) {
    console.error('合并分片失败:', error);
    return false;
  }
}

// 生成唯一ID
function generateUniqueId() {
  return Date.now().toString(36) + Math.random().toString(36).substr(2);
}

// 使用示例
const fileInput = document.querySelector('input[type="file"]');
fileInput.addEventListener('change', (e) => {
  const file = e.target.files[0];
  if (file) {
    uploadLargeFile(file);
  }
});

服务器端处理（Node.js 示例） ​

javascript
const express = require('express');
const multer = require('multer');
const fs = require('fs');
const path = require('path');

const app = express();
const upload = multer({ dest: 'uploads/' });

// 存储分片的目录
const chunksDir = 'chunks/';
if (!fs.existsSync(chunksDir)) {
  fs.mkdirSync(chunksDir, { recursive: true });
}

// 上传分片
app.post('/upload-chunk', upload.single('file'), (req, res) => {
  const { fileId, chunkIndex, totalChunks, fileName } = req.body;
  const chunkPath = path.join(chunksDir, `${fileId}-${chunkIndex}`);
  
  // 移动分片到指定位置
  fs.renameSync(req.file.path, chunkPath);
  
  res.json({ success: true, message: `分片 ${chunkIndex} 上传成功` });
});

// 合并分片
app.post('/merge-chunks', express.json(), (req, res) => {
  const { fileId, fileName, totalChunks } = req.body;
  const outputPath = path.join('uploads/', fileName);
  
  // 创建输出文件
  const output = fs.createWriteStream(outputPath);
  
  let chunkIndex = 0;
  
  function mergeNextChunk() {
    const chunkPath = path.join(chunksDir, `${fileId}-${chunkIndex}`);
    
    if (chunkIndex < totalChunks) {
      if (fs.existsSync(chunkPath)) {
        const chunk = fs.readFileSync(chunkPath);
        output.write(chunk);
        fs.unlinkSync(chunkPath); // 删除已处理的分片
        chunkIndex++;
        mergeNextChunk();
      } else {
        res.status(400).json({ success: false, message: `分片 ${chunkIndex} 不存在` });
        output.end();
      }
    } else {
      output.end();
      res.json({ success: true, message: '文件合并成功', filePath: outputPath });
    }
  }
  
  mergeNextChunk();
});

app.listen(3000, () => {
  console.log('服务器运行在 http://localhost:3000');
});

3. 断点续传 ​

原理 ​

  1. 上传前检查文件是否已存在部分上传
  2. 获取已上传的分片信息
  3. 只上传未完成的分片

实现代码 ​

javascript
async function uploadWithResume(file) {
  const chunkSize = 1024 * 1024; // 1MB 分片
  const totalChunks = Math.ceil(file.size / chunkSize);
  const fileId = generateFileId(file); // 基于文件内容生成唯一ID
  
  try {
    // 检查是否已存在上传记录
    const response = await fetch('/check-upload', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        fileId,
        fileName: file.name,
        fileSize: file.size
      })
    });
    
    const { uploadedChunks = [] } = await response.json();
    console.log(`已上传分片: ${uploadedChunks.length}/${totalChunks}`);
    
    // 上传未完成的分片
    for (let i = 0; i < totalChunks; i++) {
      // 跳过已上传的分片
      if (uploadedChunks.includes(i)) {
        console.log(`分片 ${i + 1}/${totalChunks} 已上传，跳过`);
        continue;
      }
      
      const start = i * chunkSize;
      const end = Math.min(start + chunkSize, file.size);
      const chunk = file.slice(start, end);
      
      const formData = new FormData();
      formData.append('file', chunk);
      formData.append('fileId', fileId);
      formData.append('chunkIndex', i);
      formData.append('totalChunks', totalChunks);
      formData.append('fileName', file.name);
      
      const uploadResponse = await fetch('/upload-chunk', {
        method: 'POST',
        body: formData
      });
      
      if (!uploadResponse.ok) {
        throw new Error(`上传分片 ${i} 失败`);
      }
      
      console.log(`分片 ${i + 1}/${totalChunks} 上传成功`);
    }
    
    // 所有分片上传完成后，通知服务器合并
    const mergeResponse = await fetch('/merge-chunks', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        fileId,
        fileName: file.name,
        totalChunks
      })
    });
    
    if (mergeResponse.ok) {
      const result = await mergeResponse.json();
      console.log('文件上传完成:', result);
      return true;
    } else {
      throw new Error('合并分片失败');
    }
  } catch (error) {
    console.error('上传失败:', error);
    return false;
  }
}

// 基于文件内容生成唯一ID
function generateFileId(file) {
  return new Promise((resolve) => {
    const reader = new FileReader();
    reader.onload = (e) => {
      const arrayBuffer = e.target.result;
      const hash = calculateHash(arrayBuffer); // 简单的哈希计算
      resolve(hash);
    };
    reader.readAsArrayBuffer(file);
  });
}

// 简单的哈希计算函数
function calculateHash(buffer) {
  let hash = 0;
  const view = new DataView(buffer);
  for (let i = 0; i < view.byteLength; i++) {
    hash = ((hash << 5) - hash) + view.getUint8(i);
    hash = hash & hash; // 转换为32位整数
  }
  return hash.toString(36);
}

// 检查上传状态
app.post('/check-upload', express.json(), (req, res) => {
  const { fileId } = req.body;
  const chunksDir = path.join('chunks/', fileId);
  
  if (fs.existsSync(chunksDir)) {
    // 获取已上传的分片
    const uploadedChunks = [];
    const files = fs.readdirSync(chunksDir);
    
    files.forEach(file => {
      const match = file.match(/(\d+)$/);
      if (match) {
        uploadedChunks.push(parseInt(match[1]));
      }
    });
    
    res.json({ success: true, uploadedChunks });
  } else {
    res.json({ success: true, uploadedChunks: [] });
  }
});

app.post('/upload-chunk', upload.single('file'), (req, res) => {
  const { fileId, chunkIndex, totalChunks, fileName } = req.body;
  const chunksDir = path.join('chunks/', fileId);
  
  // 创建分片目录
  if (!fs.existsSync(chunksDir)) {
    fs.mkdirSync(chunksDir, { recursive: true });
  }
  
  const chunkPath = path.join(chunksDir, chunkIndex);
  
  // 移动分片到指定位置
  fs.renameSync(req.file.path, chunkPath);
  
  res.json({ success: true, message: `分片 ${chunkIndex} 上传成功` });
});

// 修改合并分片处理
app.post('/merge-chunks', express.json(), (req, res) => {
  const { fileId, fileName, totalChunks } = req.body;
  const chunksDir = path.join('chunks/', fileId);
  const outputPath = path.join('uploads/', fileName);
  
  // 创建输出文件
  const output = fs.createWriteStream(outputPath);
  
  let chunkIndex = 0;
  
  function mergeNextChunk() {
    const chunkPath = path.join(chunksDir, chunkIndex.toString());
    
    if (chunkIndex < totalChunks) {
      if (fs.existsSync(chunkPath)) {
        const chunk = fs.readFileSync(chunkPath);
        output.write(chunk);
        chunkIndex++;
        mergeNextChunk();
      } else {
        res.status(400).json({ success: false, message: `分片 ${chunkIndex} 不存在` });
        output.end();
      }
    } else {
      output.end();
      // 删除分片目录
      fs.rmSync(chunksDir, { recursive: true, force: true });
      res.json({ success: true, message: '文件合并成功', filePath: outputPath });
    }
  }
  
  mergeNextChunk();
});

async function uploadWithProgress(file) {
  const chunkSize = 1024 * 1024; // 1MB 分片
  const totalChunks = Math.ceil(file.size / chunkSize);
  const fileId = generateUniqueId();
  let uploadedBytes = 0;
  
  // 获取进度条元素
  const progressBar = document.querySelector('.progress-bar');
  const progressText = document.querySelector('.progress-text');
  
  for (let i = 0; i < totalChunks; i++) {
    const start = i * chunkSize;
    const end = Math.min(start + chunkSize, file.size);
    const chunk = file.slice(start, end);
    
    const formData = new FormData();
    formData.append('file', chunk);
    formData.append('fileId', fileId);
    formData.append('chunkIndex', i);

    formData.append('fileName', file.name);

      const response = await fetch('/upload-chunk', {
        method: 'POST',
        body: formData
      });
      
      if (!response.ok) {
        throw new Error(`上传分片 ${i} 失败`);
      }
      
      // 更新进度
      uploadedBytes += chunk.size;
      const progress = Math.round((uploadedBytes / file.size) * 100);
      progressBar.style.width = `${progress}%`;
      progressText.textContent = `${progress}%`;
      
      console.log(`分片 ${i + 1}/${totalChunks} 上传成功`);
    } catch (error) {
      console.error('上传分片失败:', error);
      return false;
    }
  }
  
  // 合并分片
  try {
    const response = await fetch('/merge-chunks', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        fileId,
        fileName: file.name,
        totalChunks
      })
    });
    
    if (response.ok) {
      const result = await response.json();
      console.log('文件上传完成:', result);
      progressText.textContent = '上传完成';
      return true;
    } else {
      throw new Error('合并分片失败');
    }
  } catch (error) {
    console.error('合并分片失败:', error);
    return false;
  }
}

1. 并行上传 ​

javascript
async function uploadInParallel(file) {
  const chunkSize = 1024 * 1024; // 1MB 分片
  const totalChunks = Math.ceil(file.size / chunkSize);
  const fileId = generateUniqueId();
  const maxParallel = 3; // 最大并行上传数
  
  let activeUploads = 0;
  let currentChunk = 0;
  const uploadPromises = [];
  
  async function processChunk() {
    if (currentChunk >= totalChunks) return;
    
    activeUploads++;
    const chunkIndex = currentChunk++;
    
    const start = chunkIndex * chunkSize;
    const end = Math.min(start + chunkSize, file.size);
    const chunk = file.slice(start, end);
    
    const formData = new FormData();
    formData.append('file', chunk);
    formData.append('fileId', fileId);
    formData.append('chunkIndex', chunkIndex);
    formData.append('totalChunks', totalChunks);
    formData.append('fileName', file.name);
    
    try {
      const response = await fetch('/upload-chunk', {
        method: 'POST',
        body: formData
      });
      
      if (!response.ok) {
        throw new Error(`上传分片 ${chunkIndex} 失败`);
      }
      
      console.log(`分片 ${chunkIndex + 1}/${totalChunks} 上传成功`);
    } catch (error) {
      console.error('上传分片失败:', error);
      throw error;
    } finally {
      activeUploads--;
      // 继续处理下一个分片
      processChunk();
    }
  }
  
  // 启动初始并行上传
  for (let i = 0; i < maxParallel && i < totalChunks; i++) {
    uploadPromises.push(processChunk());
  }
  
  // 等待所有上传完成
  await Promise.all(uploadPromises);
  
  // 合并分片
  const response = await fetch('/merge-chunks', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      fileId,
      fileName: file.name,
      totalChunks
    })
  });
  
  if (response.ok) {
    const result = await response.json();
    console.log('文件上传完成:', result);
    return true;
  } else {
    throw new Error('合并分片失败');
  }
}

2. 错误重试机制 ​

javascript
async function uploadWithRetry(file, maxRetries = 3) {
  const chunkSize = 1024 * 1024; // 1MB 分片
  const totalChunks = Math.ceil(file.size / chunkSize);
  const fileId = generateUniqueId();
  
  for (let i = 0; i < totalChunks; i++) {
    let retries = 0;
    let uploaded = false;
    
    while (!uploaded && retries < maxRetries) {
      try {
        const start = i * chunkSize;
        const end = Math.min(start + chunkSize, file.size);
        const chunk = file.slice(start, end);
        
        const formData = new FormData();
        formData.append('file', chunk);
        formData.append('fileId', fileId);
        formData.append('chunkIndex', i);
        formData.append('totalChunks', totalChunks);
        formData.append('fileName', file.name);
        
        const response = await fetch('/upload-chunk', {
          method: 'POST',
          body: formData
        });
        
        if (response.ok) {
          console.log(`分片 ${i + 1}/${totalChunks} 上传成功`);
          uploaded = true;
        } else {
          throw new Error(`上传分片 ${i} 失败`);
        }
      } catch (error) {
        retries++;
        console.warn(`分片 ${i} 上传失败，正在重试 ${retries}/${maxRetries}...`);
        // 延迟重试
        await new Promise(resolve => setTimeout(resolve, 1000 * retries));
      }
    }
    
    if (!uploaded) {
      console.error(`分片 ${i} 上传失败，已达到最大重试次数`);
      return false;
    }
  }
  
  // 合并分片
  try {
    const response = await fetch('/merge-chunks', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        fileId,
        fileName: file.name,
        totalChunks
      })
    });
    
    if (response.ok) {
      const result = await response.json();
      console.log('文件上传完成:', result);
      return true;
    } else {
      throw new Error('合并分片失败');
    }
  } catch (error) {
    console.error('合并分片失败:', error);
    return false;
  }
}

      <p v-if="!selectedFile">拖放文件到此处或点击选择文件</p>
      <p v-else>{{ selectedFile.name }} ({{ formatFileSize(selectedFile.size) }})</p>
    </div>
    
    <div v-if="uploading" class="progress-container">
      <div class="progress-bar" :style="{ width: `${progress}%` }"></div>
      <span class="progress-text">{{ progress }}%</span>
    </div>
    
    <div class="buttons">
      <button 
        v-if="selectedFile && !uploading" 
        @click="startUpload"
        class="upload-btn"
      >
        开始上传
      </button>
      <button 
        v-if="uploading" 
        @click="cancelUpload"
        class="cancel-btn"
      >
        取消上传
      </button>
    </div>
    
    <div v-if="uploadStatus" class="status-message" :class="uploadStatus.type">
      {{ uploadStatus.message }}
    </div>
  </div>

<script setup>
import { ref, computed } from 'vue';

const fileInput = ref(null);
const selectedFile = ref(null);
const uploading = ref(false);
const progress = ref(0);
const uploadStatus = ref(null);
const uploadAbortController = ref(null);

// 处理文件选择
const handleFileSelect = (e) => {
  const files = e.target.files;
  if (files.length > 0) {
    selectedFile.value = files[0];
  }
};

// 触发文件输入
const triggerFileInput = () => {
  fileInput.value.click();
};

// 处理拖放
const handleDrop = (e) => {
  e.preventDefault();
  const files = e.dataTransfer.files;
  if (files.length > 0) {
    selectedFile.value = files[0];
  }
};

// 格式化文件大小
const formatFileSize = (bytes) => {
  if (bytes === 0) return '0 Bytes';
  const k = 1024;
  const sizes = ['Bytes', 'KB', 'MB', 'GB'];
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
};

// 开始上传
const startUpload = async () => {
  if (!selectedFile.value) return;
  
  uploading.value = true;
  progress.value = 0;
  uploadStatus.value = null;
  uploadAbortController.value = new AbortController();
  
  try {
    const success = await uploadFile(selectedFile.value);
    if (success) {
      uploadStatus.value = {
        type: 'success',
        message: '文件上传成功'
      };
    } else {
      uploadStatus.value = {
        type: 'error',
        message: '文件上传失败'
      };
    }
  } catch (error) {
    if (error.name !== 'AbortError') {
      uploadStatus.value = {
        type: 'error',
        message: '上传过程中发生错误: ' + error.message
      };
    }
  } finally {
    uploading.value = false;
  }
};

// 取消上传
const cancelUpload = () => {
  if (uploadAbortController.value) {
    uploadAbortController.value.abort();
    uploadStatus.value = {
      type: 'info',
      message: '上传已取消'
    };
    uploading.value = false;
  }
};

// 上传文件
const uploadFile = async (file) => {
  const chunkSize = 1024 * 1024; // 1MB 分片
  const totalChunks = Math.ceil(file.size / chunkSize);
  const fileId = generateUniqueId();
  let uploadedBytes = 0;
  
  for (let i = 0; i < totalChunks; i++) {
    if (uploadAbortController.value.signal.aborted) {
      throw new Error('AbortError');
    }
    
    const start = i * chunkSize;
    const end = Math.min(start + chunkSize, file.size);
    const chunk = file.slice(start, end);
    
    const formData = new FormData();
    formData.append('file', chunk);
    formData.append('fileId', fileId);
    formData.append('chunkIndex', i);
    formData.append('totalChunks', totalChunks);
    formData.append('fileName', file.name);
    
    try {
      const response = await fetch('/upload-chunk', {
        method: 'POST',
        body: formData,
        signal: uploadAbortController.value.signal
      });
      
      if (!response.ok) {
        throw new Error(`上传分片 ${i} 失败`);
      }
      
      // 更新进度
      uploadedBytes += chunk.size;
      progress.value = Math.round((uploadedBytes / file.size) * 100);
    } catch (error) {
      throw error;
    }
  }
  
  // 合并分片
  try {
    const response = await fetch('/merge-chunks', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        fileId,
        fileName: file.name,
        totalChunks
      }),
      signal: uploadAbortController.value.signal
    });
    
    if (response.ok) {
      return true;
    } else {
      throw new Error('合并分片失败');
    }
  } catch (error) {
    throw error;
  }
};

// 生成唯一ID
const generateUniqueId = () => {
  return Date.now().toString(36) + Math.random().toString(36).substr(2);
};
</script>

async function downloadLargeFile(url, filename) {
  const response = await fetch(url);
  const reader = response.body.getReader();
  const contentLength = Number(response.headers.get('content-length'));
  
  let receivedLength = 0;
  const chunks = [];
  
  while (true) {
    const { done, value } = await reader.read();
    
    if (done) {
      break;
    }
    
    chunks.push(value);
    receivedLength += value.length;
    
    // 计算并显示下载进度
    const progress = Math.round((receivedLength / contentLength) * 100);
    console.log(`下载进度: ${progress}%`);
  }
  
  // 合并所有块
  const chunksAll = new Uint8Array(receivedLength);
  let position = 0;
  for (const chunk of chunks) {
    chunksAll.set(chunk, position);
    position += chunk.length;
  }
  
  // 创建 Blob 并下载
  const blob = new Blob([chunksAll]);
  const url = URL.createObjectURL(blob);
  
  const link = document.createElement('a');
  link.href = url;
  link.download = filename;
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
  
  setTimeout(() => URL.revokeObjectURL(url), 100);
}

8.3 断点下载 ​

原理 ​

  1. 发送 HEAD 请求获取文件大小
  2. 检查本地已下载的文件大小
  3. 使用 Range 请求头从断点处继续下载
  4. 将新下载的内容追加到本地文件

实现代码 ​

javascript
async function resumeDownload(url, filename) {
  // 检查本地是否已有部分下载的文件
  let localFile = null;
  let downloadedSize = 0;
  
  try {
    // 这里使用 localStorage 模拟本地存储，实际应用中可能需要使用 IndexedDB 或 File System API
    const savedProgress = localStorage.getItem(`download_${filename}`);
    if (savedProgress) {
      const progress = JSON.parse(savedProgress);
      downloadedSize = progress.downloadedSize;
      localFile = progress.data;
    }
  } catch (error) {
    console.error('读取本地下载进度失败:', error);
  }
  
  // 获取文件总大小
  const headResponse = await fetch(url, { method: 'HEAD' });
  const contentLength = Number(headResponse.headers.get('content-length'));
  
  if (downloadedSize >= contentLength) {
    console.log('文件已完全下载');
    return;
  }
  
  // 发送 Range 请求
  const response = await fetch(url, {
    headers: {
      'Range': `bytes=${downloadedSize}-${contentLength - 1}`
    }
  });
  
  if (!response.ok) {
    throw new Error(`下载失败: ${response.status}`);
  }
  
  const reader = response.body.getReader();
  const chunks = localFile ? [new Uint8Array(localFile)] : [];
  
  while (true) {
    const { done, value } = await reader.read();
    
    if (done) {
      break;
    }
    
    chunks.push(value);
    downloadedSize += value.length;
    
    // 计算并显示下载进度
    const progress = Math.round((downloadedSize / contentLength) * 100);
    console.log(`下载进度: ${progress}%`);
    
    // 保存下载进度
    try {
      // 合并当前所有块以保存进度
      const totalArray = new Uint8Array(downloadedSize);
      let position = 0;
      for (const chunk of chunks) {
        totalArray.set(chunk, position);
        position += chunk.length;
      }
      
      localStorage.setItem(`download_${filename}`, JSON.stringify({
        downloadedSize,
        data: Array.from(totalArray)
      }));
    } catch (error) {
      console.error('保存下载进度失败:', error);
    }
  }
  
  // 合并所有块
  const chunksAll = new Uint8Array(downloadedSize);
  let position = 0;
  for (const chunk of chunks) {
    chunksAll.set(chunk, position);
    position += chunk.length;
  }
  
  // 创建 Blob 并下载
  const blob = new Blob([chunksAll]);
  const downloadUrl = URL.createObjectURL(blob);
  
  const link = document.createElement('a');
  link.href = downloadUrl;
  link.download = filename;
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
  
  setTimeout(() => URL.revokeObjectURL(downloadUrl), 100);
  
  // 清除下载进度记录
  localStorage.removeItem(`download_${filename}`);
}

// 使用示例
resumeDownload('https://example.com/large-file.zip', 'large-file.zip');

// Node.js Express 示例
app.get('/download', (req, res) => {
  const filePath = path.join(__dirname, 'files', 'large-file.zip');
  const fileSize = fs.statSync(filePath).size;
  
  // 处理 Range 请求
  const range = req.headers.range;
  if (range) {
    const parts = range.replace(/bytes=/, '').split('-');
    const start = parseInt(parts[0], 10);
    const end = parts[1] ? parseInt(parts[1], 10) : fileSize - 1;
    
    const chunkSize = end - start + 1;
    const file = fs.createReadStream(filePath, { start, end });
    
    const headers = {
      'Content-Range': `bytes ${start}-${end}/${fileSize}`,
      'Accept-Ranges': 'bytes',
      'Content-Length': chunkSize,
      'Content-Type': 'application/octet-stream',
      'Content-Disposition': 'attachment; filename=large-file.zip'
    };
    
    res.writeHead(206, headers);
    file.pipe(res);
  } else {
    // 完整文件下载
    const headers = {
      'Content-Length': fileSize,
      'Content-Type': 'application/octet-stream',
      'Content-Disposition': 'attachment; filename=large-file.zip'
    };
    
    res.writeHead(200, headers);
    fs.createReadStream(filePath).pipe(res);
  }
});

8.5 下载优化策略 ​

  1. 多线程下载：

    • 同时发起多个 Range 请求，分别下载文件的不同部分
    • 下载完成后合并所有部分

  2. 下载速度限制：

    • 控制下载速度，避免占用过多带宽

  3. 自动重试：

    • 当下载失败时自动重试

  4. 下载队列管理：

    • 管理多个下载任务，避免同时下载过多文件

9. 总结 ​

文件上传和下载是Web应用中常见的功能，随着文件大小的增加，传统的方式会遇到各种问题。通过以下技术，可以构建高效、可靠的文件传输系统：

上传技术 ​

  1. 分片上传：将大文件分割成小分片，逐个上传，减少单个请求的大小，提高上传成功率。

  2. 断点续传：在上传过程中记录已上传的分片，当上传中断后可以从断点处继续上传，避免重复上传。

  3. 并行上传：同时上传多个分片，提高上传速度。

  4. 错误重试：当分片上传失败时，自动重试，提高上传可靠性。

  5. 进度显示：实时显示上传进度，提升用户体验。

下载技术 ​

  1. 基本下载：直接链接下载、JavaScript 触发下载、Blob 对象下载。

  2. 大文件下载：流式下载，避免一次性加载整个文件到内存。

  3. 断点下载：从上次中断的位置继续下载，避免重复下载。

  4. 多线程下载：同时下载文件的不同部分，提高下载速度。

  5. 服务器端支持：实现 Range 请求处理，支持断点续传。

通过这些技术的组合使用，可以构建一个高效、可靠的文件传输系统，满足各种复杂场景的需求。
