Promise 实现原理深度解析 ​

什么是 Promise ​

Promise 是 JavaScript 中用于处理异步操作的对象，它表示一个异步操作的最终完成（或失败）及其结果值。

Promise 的状态 ​

Promise 有三种状态：

  • Pending（待处理）：初始状态，既不是成功也不是失败
  • Fulfilled（已完成）：操作成功完成
  • Rejected（已拒绝）：操作失败

状态一旦改变，就不能再被修改。

Promise 的基本使用 ​

javascript
const promise = new Promise((resolve, reject) => {
  // 异步操作
  setTimeout(() => {
    if (Math.random() > 0.5) {
      resolve('操作成功');
    } else {
      reject('操作失败');
    }
  }, 1000);
});

promise.then(
  (value) => console.log('成功:', value),
  (error) => console.log('失败:', error)
);

Promise 的实现原理 ​

1. 基本结构 ​

javascript
class MyPromise {
  constructor(executor) {
    // 初始状态
    this.state = 'pending';
    // 成功时的值
    this.value = undefined;
    // 失败时的原因
    this.reason = undefined;
    // 成功回调函数队列
    this.onFulfilledCallbacks = [];
    // 失败回调函数队列
    this.onRejectedCallbacks = [];

    // 定义 resolve 函数
    const resolve = (value) => {
      if (this.state === 'pending') {
        this.state = 'fulfilled';
        this.value = value;
        // 执行所有成功回调
        this.onFulfilledCallbacks.forEach(callback => callback(value));
      }
    };

    // 定义 reject 函数
    const reject = (reason) => {
      if (this.state === 'pending') {
        this.state = 'rejected';
        this.reason = reason;
        // 执行所有失败回调
        this.onRejectedCallbacks.forEach(callback => callback(reason));
      }
    };

    // 执行 executor 函数
    try {
      executor(resolve, reject);
    } catch (error) {
      reject(error);
    }
  }
}

2. then 方法的实现 ​

javascript
then(onFulfilled, onRejected) {
  // 参数默认值处理
  onFulfilled = typeof onFulfilled === 'function' ? onFulfilled : value => value;
  onRejected = typeof onRejected === 'function' ? onRejected : reason => { throw reason };

  // 创建新的 Promise 以支持链式调用
  const promise2 = new MyPromise((resolve, reject) => {
    if (this.state === 'fulfilled') {
      // 异步执行，确保 then 方法是微任务
      setTimeout(() => {
        try {
          const x = onFulfilled(this.value);
          this.resolvePromise(promise2, x, resolve, reject);
        } catch (error) {
          reject(error);
        }
      }, 0);
    }

    if (this.state === 'rejected') {
      // 异步执行，确保 then 方法是微任务
      setTimeout(() => {
        try {
          const x = onRejected(this.reason);
          this.resolvePromise(promise2, x, resolve, reject);
        } catch (error) {
          reject(error);
        }
      }, 0);
    }

    if (this.state === 'pending') {
      // 保存回调函数到队列
      this.onFulfilledCallbacks.push(value => {
        setTimeout(() => {
          try {
            const x = onFulfilled(value);
            this.resolvePromise(promise2, x, resolve, reject);
          } catch (error) {
            reject(error);
          }
        }, 0);
      });

      this.onRejectedCallbacks.push(reason => {
        setTimeout(() => {
          try {
            const x = onRejected(reason);
            this.resolvePromise(promise2, x, resolve, reject);
          } catch (error) {
            reject(error);
          }
        }, 0);
      });
    }
  });

  return promise2;
}

3. resolvePromise 方法的实现 ​

javascript
resolvePromise(promise2, x, resolve, reject) {
  // 防止循环引用
  if (promise2 === x) {
    return reject(new TypeError('Chaining cycle detected for promise'));
  }

  // 处理 x 是 Promise 的情况
  if (x instanceof MyPromise) {
    x.then(
      value => this.resolvePromise(promise2, value, resolve, reject),
      reason => reject(reason)
    );
  } else if (x !== null && (typeof x === 'object' || typeof x === 'function')) {
    // 处理 x 是对象或函数的情况
    let called = false;
    try {
      const then = x.then;
      if (typeof then === 'function') {
        // x 是 thenable 对象
        then.call(
          x,
          value => {
            if (called) return;
            called = true;
            this.resolvePromise(promise2, value, resolve, reject);
          },
          reason => {
            if (called) return;
            called = true;
            reject(reason);
          }
        );
      } else {
        // x 是普通对象
        resolve(x);
      }
    } catch (error) {
      if (called) return;
      called = true;
      reject(error);
    }
  } else {
    // x 是普通值
    resolve(x);
  }
}

Promise 的链式调用 ​

原理 ​

Promise 的链式调用是通过在 then 方法中返回一个新的 Promise 实现的。每个 then 方法都会返回一个新的 Promise，从而实现链式调用。

示例 ​

javascript
new Promise((resolve) => {
  setTimeout(() => resolve(1), 1000);
})
  .then(value => value + 1)
  .then(value => value + 1)
  .then(value => console.log(value)); // 输出: 3

all 方法 ​

javascript
static all(promises) {
  return new MyPromise((resolve, reject) => {
    const results = [];
    let count = 0;
    
    for (let i = 0; i < promises.length; i++) {
      MyPromise.resolve(promises[i]).then(
        value => {
          results[i] = value;
          count++;
          if (count === promises.length) {
            resolve(results);
          }
        },
        reason => reject(reason)
      );
    }
  });
}

static allSettled(promises) {

any 方法 ​

javascript
static any(promises) {
  return new MyPromise((resolve, reject) => {
    const errors = [];
    let count = 0;
    
    for (let i = 0; i < promises.length; i++) {
      MyPromise.resolve(promises[i]).then(
        value => resolve(value),
        reason => {
          errors[i] = reason;
          count++;
          if (count === promises.length) {
            reject(new AggregateError(errors, 'All promises were rejected'));
          }
        }
      );
    }
  });
}

Promise 的执行顺序 ​

微任务特性 ​

Promise 的 then 方法回调是微任务，会在同步代码执行完毕后、DOM 更新前执行。

示例 ​

javascript
console.log('同步代码开始');

new Promise(resolve => {
  console.log('Promise 执行器');
  resolve();
}).then(() => {
  console.log('Promise 回调');
});

console.log('同步代码结束');

// 输出顺序：
// 同步代码开始
// Promise 执行器
// 同步代码结束
// Promise 回调

Promise 与 async/await 的关系 ​

async/await 是基于 Promise 的语法糖，使得异步代码看起来更像同步代码。

javascript
// 使用 Promise
function fetchData() {
  return fetch('https://api.example.com/data')
    .then(response => response.json())
    .then(data => data);
}

// 使用 async/await
async function fetchData() {
  const response = await fetch('https://api.example.com/data');
  const data = await response.json();
  return data;
}

面试示例回答 ​

问题：请解释 Promise 的实现原理

回答：

Promise 的实现原理主要基于以下几个核心概念：

首先，Promise 有三种状态：Pending（待处理）、Fulfilled（已完成）和 Rejected（已拒绝）。状态一旦改变，就不能再被修改。

其次，Promise 的核心是一个执行器函数，在创建 Promise 时立即执行。执行器函数接收两个参数：resolve 和 reject，用于改变 Promise 的状态。

然后，Promise 通过回调队列来管理异步操作。当 Promise 的状态改变时，会执行相应的回调函数。

Promise 的 then 方法是实现链式调用的关键。它会返回一个新的 Promise，从而允许我们链式调用多个 then 方法。

在 then 方法中，会根据当前 Promise 的状态执行相应的回调函数。如果状态是 Pending，会将回调函数保存到队列中；如果状态是 Fulfilled 或 Rejected，会异步执行相应的回调函数。

Promise 还实现了 resolvePromise 方法，用于处理 then 方法返回的值，确保链式调用的正确性。

此外，Promise 的回调函数是微任务，会在同步代码执行完毕后、DOM 更新前执行，这使得 Promise 的执行顺序更加可预测。

通过这些机制，Promise 提供了一种优雅的方式来处理异步操作，避免了回调地狱，使代码更加清晰和可维护。
